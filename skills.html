// C++ DSA Solution: Efficient Graph BFS & Shortest Path
#include <iostream>
#include <vector>
#include <queue>

using namespace std;

class Graph {
    int V;
    vector<vector<int>> adj;
public:
    Graph(int v) : V(v), adj(v) {}
    
    void addEdge(int u, int v) {
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    void shortestPathBFS(int startNode) {
        vector<int> dist(V, -1);
        queue<int> q;
        
        dist[startNode] = 0;
        q.push(startNode);
        
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            
            for (int neighbor : adj[u]) {
                if (dist[neighbor] == -1) {
                    dist[neighbor] = dist[u] + 1;
                    q.push(neighbor);
                }
            }
        }
    }
};
